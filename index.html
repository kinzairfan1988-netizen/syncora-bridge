async function confirmDispatchTranslation() { if (!pendingPayload || !socket) return; // Dropdown se selected language ki sahi value read karein const langSelect = document.getElementById("modal-target-lang"); const targetLang = langSelect ? langSelect.value : "ur"; const btn = document.getElementById("btn-modal-confirm-trans"); const origText = btn.innerText; btn.innerText = "Translating..."; btn.disabled = true; let translated = ""; try { const res = await fetch("/translate", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ text: pendingPayload.content, target_lang: targetLang }) }); const data = await res.json(); translated = data.translated_text || ""; } catch (e) { console.error("Translation request error:", e); } btn.innerText = origText; btn.disabled = false; // Agar translation empty ho toh fallback if (!translated || translated.trim() === "") { translated = pendingPayload.content; } executeDispatch(pendingPayload.content, translated, pendingPayload.type); cancelDispatch(); }
